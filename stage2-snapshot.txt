===== client/src/lib/analytics.ts =====
import { apiPost } from "@/lib/api";

/*
  Privacy-first visit tracking for POST /api/analytics/track.

  How the server behaves (server/src/lib/analytics.ts):
  - It stores ONE row per visitor per day (hash of IP + date) and silently
    drops every later ping from that visitor.
  - That row can carry a `projectClicked`, but only if it is in the FIRST ping.

  So a ping on page load would "use up" the visitor's row for the day and every
  project click after it would be lost. Instead the client sends exactly ONE
  ping per page visit, at the moment it has the most to report:
    1. the visitor clicks a project link   -> ping now, with `projectClicked`
    2. otherwise, the visitor leaves/hides  -> ping then (page view only)

  Nothing is stored in the browser (no cookies, no localStorage). Tracking is
  skipped when Do Not Track / Global Privacy Control is on, and in development
  unless VITE_ENABLE_ANALYTICS=true.
*/

const TRACK_PATH = "/analytics/track";

// Loose 8-4-4-4-12 hex check. The server casts the body without validating it,
// and an invalid id would make Postgres throw *after* the day's slot is taken.
const UUID_PATTERN =
  /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;

interface TrackPayload {
  referrer?: string;
  projectClicked?: string;
}

/** The one ping for this page visit has been sent (or tracking is off). */
let pingSent = false;
let enabled: boolean | undefined;

function isEnabled(): boolean {
  if (enabled !== undefined) return enabled;

  const allowedHere =
    import.meta.env.PROD || import.meta.env.VITE_ENABLE_ANALYTICS === "true";
  const globalPrivacyControl =
    (navigator as Navigator & { globalPrivacyControl?: boolean })
      .globalPrivacyControl === true;
  const doNotTrack =
    navigator.doNotTrack === "1" || navigator.doNotTrack === "yes";

  enabled = allowedHere && !globalPrivacyControl && !doNotTrack;
  return enabled;
}

/**
 * Where the visitor came from, reduced to an origin ("https://t.co").
 * Same-origin referrers (reloads, in-site navigation) are not interesting.
 */
function readReferrer(): string | undefined {
  if (!document.referrer) return undefined;
  try {
    const { origin } = new URL(document.referrer);
    return origin === window.location.origin ? undefined : origin;
  } catch {
    return undefined;
  }
}

function sendPing(projectId?: string): void {
  if (pingSent) return;
  pingSent = true;

  const payload: TrackPayload = {};
  const referrer = readReferrer();
  if (referrer) payload.referrer = referrer;
  if (projectId && UUID_PATTERN.test(projectId)) {
    payload.projectClicked = projectId;
  }

  // Fire and forget: analytics must never affect, slow or break the page.
  apiPost(TRACK_PATH, payload, { keepalive: true }).catch(() => undefined);
}

/**
 * Registers the "visitor left" listeners. Returns a cleanup function.
 * `visibilitychange` is the reliable signal on mobile (tab switch, app switch);
 * `pagehide` covers closing/navigating away on desktop.
 */
export function startVisitTracking(): () => void {
  if (!isEnabled()) return () => undefined;

  const onVisibilityChange = () => {
    if (document.visibilityState === "hidden") sendPing();
  };
  const onPageHide = () => {
    sendPing();
  };

  document.addEventListener("visibilitychange", onVisibilityChange);
  window.addEventListener("pagehide", onPageHide);

  return () => {
    document.removeEventListener("visibilitychange", onVisibilityChange);
    window.removeEventListener("pagehide", onPageHide);
  };
}

/** Call when a visitor follows a project's live/source link. First click wins. */
export function trackProjectClick(projectId: string): void {
  if (!isEnabled()) return;
  sendPing(projectId);
}

===== client/src/stores/theme.ts =====
import { create } from "zustand";

export type Theme = "dark" | "light";

// Keep in sync with the inline script in index.html.
const STORAGE_KEY = "theme";

/** The inline script in index.html has already set data-theme before React runs. */
function readTheme(): Theme {
  return document.documentElement.dataset.theme === "light" ? "light" : "dark";
}

/** Point the browser UI color at the active --bg token (mobile address bar). */
function syncThemeColor(): void {
  const bg = getComputedStyle(document.documentElement)
    .getPropertyValue("--bg")
    .trim();
  if (!bg) return;
  document
    .querySelector('meta[name="theme-color"]')
    ?.setAttribute("content", bg);
}

function applyTheme(theme: Theme): void {
  document.documentElement.dataset.theme = theme;
  try {
    localStorage.setItem(STORAGE_KEY, theme);
  } catch {
    // Storage can be blocked (e.g. private mode); the theme still applies, it just won't persist.
  }
  syncThemeColor();
}

interface ThemeState {
  theme: Theme;
  setTheme: (theme: Theme) => void;
  toggleTheme: () => void;
}

export const useThemeStore = create<ThemeState>()((set, get) => ({
  theme: readTheme(),
  setTheme: (theme) => {
    applyTheme(theme);
    set({ theme });
  },
  toggleTheme: () => {
    get().setTheme(get().theme === "dark" ? "light" : "dark");
  },
}));

syncThemeColor();

import { useThemeStore } from "@/stores/theme";

/**
 * DARK / LIGHT segmented switch. A single native button (role="switch") so it
 * is keyboard operable; the visible labels are decorative for assistive tech.
 */
export function ThemeToggle() {
  const theme = useThemeStore((state) => state.theme);
  const toggleTheme = useThemeStore((state) => state.toggleTheme);
  const isLight = theme === "light";

  return (
    <button
      type="button"
      role="switch"
      aria-checked={isLight}
      aria-label="Light theme"
      onClick={toggleTheme}
      className="relative grid h-9.25 w-36.75 grid-cols-2 items-center rounded-full bg-surface-raised p-1 text-micro font-bold tracking-label uppercase"
    >
      <span
        aria-hidden
        className={cn(
          "absolute top-1 bottom-1 left-1 w-[calc(50%-0.25rem)] rounded-full bg-accent transition-transform duration-300 motion-reduce:transition-none",
          isLight && "translate-x-full",
        )}
      />
      <span
        aria-hidden
        className={cn(
          "relative z-10 text-center transition-colors",
          isLight ? "text-fg-muted" : "text-accent-fg",

        Dark
      </span>
      <span
        aria-hidden
        className={cn(

  );

import { 
    useEffect, useRef, 
    type KeyboardEvent, 
    type RefObject }    from "react";
import { X }            from "lucide-react";
import { NAV_LINKS }    from "@/content/home";
import { Container }    from "@/components/ui/container";
import { ThemeToggle }  from "@/components/ui/theme-toggle";
import { Wordmark }     from "@/components/ui/wordmark";
import { useLockBodyScroll } from "@/hooks/use-lock-body-scroll";
import { cn } from "@/lib/cn";

interface MobileMenuProps {
  id: string;
  open: boolean;
  /** `returnFocus` is false when a link was chosen (focus should follow the navigation). */
  onClose: (returnFocus: boolean) => void;
  /** Receives focus when the menu opens. */
  closeButtonRef: RefObject<HTMLButtonElement | null>;
}

const FOCUSABLE = "a[href], button:not([disabled])";

export function MobileMenu({
  id,
  open,
  onClose,
  closeButtonRef,
}: MobileMenuProps) {
  const panelRef = useRef<HTMLDivElement>(null);

  useLockBodyScroll(open);

  // Move focus into the dialog when it opens.
  useEffect(() => {
    if (open) closeButtonRef.current?.focus();
  }, [open, closeButtonRef]);

  // Escape closes the menu.
  useEffect(() => {
    if (!open) return;
    const onKeyDown = (event: globalThis.KeyboardEvent) => {
      if (event.key === "Escape") onClose(true);
    };
    document.addEventListener("keydown", onKeyDown);
    return () => document.removeEventListener("keydown", onKeyDown);
  }, [open, onClose]);

  // Keep Tab / Shift+Tab inside the dialog while it is open.
  const trapFocus = (event: KeyboardEvent<HTMLDivElement>) => {
    if (event.key !== "Tab") return;
    const items = panelRef.current?.querySelectorAll<HTMLElement>(FOCUSABLE);
    const first = items?.[0];
    const last = items?.[items.length - 1];
    if (!first || !last) return;

    if (event.shiftKey && document.activeElement === first) {
      event.preventDefault();
      last.focus();
    } else if (!event.shiftKey && document.activeElement === last) {
      event.preventDefault();
      first.focus();
    }
  };

  return (
    <div
      ref={panelRef}
      id={id}
      role="dialog"
      aria-modal="true"
      aria-label="Site menu"
      inert={!open}
      onKeyDown={trapFocus}
      className={cn(
        "fixed inset-0 z-50 flex flex-col overscroll-contain bg-bg motion-reduce:transition-none md:hidden",
        // Opening shows the menu immediately (so it can take focus) and fades in;
        // closing fades out first and only then switches to `visibility: hidden`.
        open
          ? "visible translate-y-0 opacity-100 [transition:opacity_300ms_ease-out,transform_300ms_ease-out,visibility_0s_linear_0s]"
          : "invisible -translate-y-2 opacity-0 [transition:opacity_300ms_ease-out,transform_300ms_ease-out,visibility_0s_linear_300ms]",
      )}
    >
      <div className="border-b border-border">
        <Container className="flex h-19.25 items-center justify-between">
          <Wordmark />
          <button
            ref={closeButtonRef}
            type="button"
            aria-label="Close menu"
            onClick={() => {
              onClose(true);
            }}
            className="grid size-11 place-items-center rounded-full bg-surface-raised text-fg transition-colors hover:text-accent-text"
          >
            <X aria-hidden size={20} />
          </button>
        </Container>
      </div>

      <nav aria-label="Mobile" className="min-h-0 flex-1 overflow-y-auto pt-6">
        <Container>
          <ul>
            {NAV_LINKS.map((link, index) => (
              <li
                key={link.href}
                style={{
                  transitionDelay: open ? `${90 + index * 55}ms` : "0ms",
                }}
                className={cn(
                  "border-b border-border transition-[opacity,transform] duration-300 ease-out motion-reduce:transition-none",
                  open
                    ? "translate-y-0 opacity-100"
                    : "translate-y-3 opacity-0",
                )}
              >
                <a
                  href={link.href}
                  onClick={() => {
                    onClose(false);
                  }}
                  className="flex items-baseline gap-5 py-5 font-display text-[2.75rem] leading-none font-bold text-fg uppercase transition-colors hover:text-accent-text focus-visible:text-accent-text active:text-accent-text"
                >
                  <span className="w-6 text-label font-medium tracking-label text-accent-text">
                    {String(index + 1).padStart(2, "0")}
                  </span>
                  {link.label}
                </a>
              </li>
            ))}
          </ul>
        </Container>
      </nav>

      {/* Wrapper keeps the container full-width inside this flex column. */}
      <div className="w-full">
        <Container className="flex items-center justify-between pt-6 pb-8">
          <span className="text-micro tracking-label text-fg-subtle uppercase">
            Theme
          </span>
          <ThemeToggle />
        </Container>
      </div>
    </div>
  );
}

===== client/src/components/ui/link-button.tsx =====
import type { ComponentPropsWithoutRef, ReactNode } from "react";
import { buttonClasses, type ButtonVariant } from "./button-styles";

interface LinkButtonProps extends Omit<
  ComponentPropsWithoutRef<"a">,
  "href" | "className"
> {
  href: string;
  variant?: ButtonVariant;
  /** Opens in a new tab with a safe `rel`, and announces that to screen readers. */
  external?: boolean;
  icon?: ReactNode;
  className?: string;
}

/** Anchor styled as a button. Always renders a real, navigable link. */
export function LinkButton({
  href,
  variant = "primary",
  external = false,
  icon,
  className,
  children,
  ...props
}: LinkButtonProps) {
  return (
    <a
      href={href}
      className={buttonClasses(variant, className)}
      {...(external ? { target: "_blank", rel: "noopener noreferrer" } : {})}
      {...props}
    >
      <span>
        {children}
        {external ? (
          <span className="sr-only"> (opens in a new tab)</span>
        ) : null}
      </span>
      {icon ? <span aria-hidden>{icon}</span> : null}
    </a>
  );
}
